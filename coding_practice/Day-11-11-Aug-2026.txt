# provider "aws" tells Terraform that we are going to use AWS.
# Create the AWS resources in US East (N. Virginia).
provider "aws" {
  region = "us-east-1"
}
# I want to create an EC2 instance.
# There are two important parts:
# aws_instance  → Resource type
#PathnexEC2    → Terraform's local name for this resource

# It defines the operating system/image from which the EC2 instance will be created.
ami = "ami-0abcd1234abcd1234"
# r5.2xlarge is a memory-optimized EC2 instance type.
instance_type = "r5.2xlarge"
# Tags are key-value pairs used to identify and organize AWS resources.
#| Tag           | Value            | Purpose                              |
#| ------------- | ---------------- | ------------------------------------ |
#| `Name`        | `Pathnex-Server` | EC2 server name                      |
#| `Environment` | `Training`       | Shows this is a training environment |
#| `Owner`       | `PathnexStudent` | Identifies the owner                 |

tags = {
  Name        = "Pathnex-Server"
  Environment = "Training"
  Owner       = "PathnexStudent"
}

## 🔹 Ansible — Create User

---
- name: Create Pathnex user
  hosts: all
  become: yes

  tasks:

      user:
        name: pathnex-admin
        shell: /bin/bash

become: yes
# Contains the actual work Ansible needs to perform.
tasks:
# Just a description displayed when the playbook runs.
- name: Create user pathnex-admin
# This is Ansible's user-management module. It manages Linux user accounts.
user:

name: pathnex-admin
# Sets the user's login shell to Bash.
